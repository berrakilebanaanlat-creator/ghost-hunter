 * - 10-50sn rastgele aralıklarla Türkçe kelime seslendirme (expo-speech)
 * - Arka plan cızırtı sesi (static_loop.mp3)
 * - Sinyal tespit anında kelime + ses efekti

 * - ScrollView ile tüm içerik görünür
 *
 * Gerekli ses dosyaları (assets/sounds/):
 *   - static_loop.mp3 (arka plan cızırtı - loop)

 * Bağımlılık: expo-speech (pnpm add expo-speech)

 * Dosya: app/(tabs)/frequency-scanner.tsx

  View,
  Text,
  ScrollView,
  Pressable,

import * as Speech from "expo-speech";
import * as Haptics from "expo-haptics";

// ─── TYPES ───────────────────────────────────────────────────────────────────

type Band = "AM" | "FM";
type ScanState = "idle" | "scanning" | "detected";

interface Detection {
  id: string;
  word: string;
  frequency: number;
  band: Band;
  strength: number;
  timestamp: number;
}

// ─── TÜRKÇE KELİME BANKASI (Frekans Tarayıcı için) ─────────────────────────
// Paranormal/ürkütücü kelimeler - radyo frekansından geliyormuş gibi
const SPIRIT_WORDS: string[] = [
  // Kısa, ürkütücü kelimeler
  "buradayım", "gel", "gitme", "dur", "bak", "arkanda",
  "görüyorum", "duyuyorum", "yardım", "karanlık", "soğuk",
  "kim", "neden", "nerede", "sessiz", "dinle",
  "kapı", "ayna", "gölge", "ruh", "ışık",

  // Daha uzun, anlaşılması zor olanlar (fısıltı efekti)
  "seni...izliyorum", "burası...soğuk", "yalnız...değilsin",
  "duy...beni", "gel...yanıma", "kork...ma",
  "uyan", "hatırla", "affet", "özledim",

  // Tek heceli, radyo parazitinden çıkmış gibi
  "ah", "of", "şş", "hey", "hah",
  "tık", "çat", "vız", "hış", "pat",
];

// ─── CONSTANTS ───────────────────────────────────────────────────────────────

const { width: SCREEN_WIDTH } = Dimensions.get("window");

const BANDS = {
  AM: { min: 530, max: 1700, unit: "kHz", step: 10 },
  FM: { min: 87.5, max: 108.0, unit: "MHz", step: 0.1 },
};

// Rastgele aralıklar (ms) - 10sn ile 50sn arası
const DETECTION_INTERVALS = [
  10000, 12000, 15000, 18000, 22000, 25000, 28000,
  30000, 35000, 38000, 40000, 42000, 45000, 48000, 50000,
];

// TTS ses ayarları - farklı "ruh" sesleri
const VOICE_PROFILES = [
  { pitch: 0.3, rate: 0.4 },   // Çok derin, yavaş
  { pitch: 0.5, rate: 0.6 },   // Derin
  { pitch: 0.7, rate: 0.8 },   // Normal-derin
  { pitch: 1.2, rate: 1.3 },   // Hızlı, tiz
  { pitch: 1.5, rate: 0.5 },   // Çok tiz, yavaş (ürkütücü)
  { pitch: 0.4, rate: 1.0 },   // Derin, normal hız
  { pitch: 0.8, rate: 0.3 },   // Çok yavaş fısıltı
  { pitch: 1.0, rate: 0.7 },   // Normal pitch, yavaş
];

// ─── COMPONENT ───────────────────────────────────────────────────────────────

export default function FrequencyScanner() {
  // State
  const [band, setBand] = useState<Band>("FM");
  const [frequency, setFrequency] = useState<number>(BANDS.FM.min);
  const [scanState, setScanState] = useState<ScanState>("idle");
  const [signalStrength, setSignalStrength] = useState<number>(0);
  const [detections, setDetections] = useState<Detection[]>([]);
  const [currentWord, setCurrentWord] = useState<string>("");
  const [scanDirection, setScanDirection] = useState<1 | -1>(1);
  const [elapsedTime, setElapsedTime] = useState<number>(0);

  // Refs
  const scanIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const timerIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const detectionTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const isScanningRef = useRef(false);

  // Animations
  const pulseAnim = useRef(new Animated.Value(1)).current;
  const glowAnim = useRef(new Animated.Value(0)).current;
  const noiseAnim = useRef(new Animated.Value(0.3)).current;
  const wordFadeAnim = useRef(new Animated.Value(0)).current;

  // Audio - arka plan cızırtı
  const staticPlayer = useAudioPlayer(
    require("@/assets/sounds/static_loop.mp3")
  );

  // ─── AUDIO SETUP ────────────────────────────────────────────────────────────

  useEffect(() => {
    setAudioModeAsync({ playsInSilentModeIOS: true });
    return () => {
      staticPlayer.release();
      Speech.stop();
    };
  }, []);

  // ─── ANIMATIONS ─────────────────────────────────────────────────────────────

  useEffect(() => {
    if (scanState === "scanning" || scanState === "detected") {
      // Pulse
      Animated.loop(
        Animated.sequence([
          Animated.timing(pulseAnim, {
            toValue: 1.02,
            duration: 1200,
            easing: Easing.inOut(Easing.ease),
            useNativeDriver: true,
          }),
          Animated.timing(pulseAnim, {
            toValue: 1,
            duration: 1200,
            easing: Easing.inOut(Easing.ease),
            useNativeDriver: true,
          }),
        ])
      ).start();

      // Noise flicker
      Animated.loop(
        Animated.sequence([
          Animated.timing(noiseAnim, {
            toValue: 0.7,
            duration: 80 + Math.random() * 150,
            useNativeDriver: true,
          }),
          Animated.timing(noiseAnim, {
            toValue: 0.25,
            duration: 80 + Math.random() * 150,
            useNativeDriver: true,
          }),
        ])
      ).start();
    } else {
      pulseAnim.setValue(1);
      noiseAnim.setValue(0.3);
    }
  }, [scanState]);

  // ─── SPEECH (TTS) - Türkçe kelime seslendirme ─────────────────────────────

  const speakWord = useCallback((word: string) => {
    const profile = VOICE_PROFILES[Math.floor(Math.random() * VOICE_PROFILES.length)];

    Speech.speak(word, {
      language: "tr-TR",
      pitch: profile.pitch,
      rate: profile.rate,
      volume: 0.8 + Math.random() * 0.2,
      onDone: () => {
        // Kelime bittikten sonra fade out
        Animated.timing(wordFadeAnim, {
          toValue: 0,
          duration: 2000,
          useNativeDriver: true,
        }).start(() => {
          setCurrentWord("");
        });
      },
    });

    // Kelime göründüğünde fade in
    setCurrentWord(word);
    Animated.timing(wordFadeAnim, {
      toValue: 1,
      duration: 300,
      useNativeDriver: true,
    }).start();
  }, []);

  // ─── DETECTION LOGIC ────────────────────────────────────────────────────────

  const getRandomInterval = useCallback(() => {
    return DETECTION_INTERVALS[Math.floor(Math.random() * DETECTION_INTERVALS.length)];
  }, []);

  const triggerDetection = useCallback(() => {
    if (!isScanningRef.current) return;

    // Rastgele kelime seç
    const word = SPIRIT_WORDS[Math.floor(Math.random() * SPIRIT_WORDS.length)];
    const strength = 35 + Math.floor(Math.random() * 65); // 35-100

    // Frekansı bir "paranormal" değere kilitle
    const paranormalFreq = band === "FM"
      ? +(87.5 + Math.random() * 20.5).toFixed(1)
      : Math.round(530 + Math.random() * 1170);

    setFrequency(paranormalFreq);
    setSignalStrength(strength);
    setScanState("detected");

    // Haptic
    if (Platform.OS !== "web") {
      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning);
    }

    // Kelimeyi seslendir
    speakWord(word);

    // Glow animation
    Animated.sequence([
      Animated.timing(glowAnim, {
        toValue: 1,
        duration: 200,
        useNativeDriver: true,
      }),
      Animated.timing(glowAnim, {
        toValue: 0.4,
        duration: 300,
        useNativeDriver: true,
      }),
      Animated.timing(glowAnim, {
        toValue: 0.8,
        duration: 200,
        useNativeDriver: true,
      }),
      Animated.timing(glowAnim, {
        toValue: 0,
        duration: 2000,
        useNativeDriver: true,
      }),
    ]).start();

    // Geçmişe ekle
    const detection: Detection = {
      id: Date.now().toString(),
      word,
      frequency: paranormalFreq,
      band,
      strength,
      timestamp: Date.now(),
    };
    setDetections((prev) => [detection, ...prev].slice(0, 6));

    // 4sn sonra taramaya devam
    setTimeout(() => {
      if (isScanningRef.current) {
        setScanState("scanning");
        setSignalStrength(0);
        scheduleNextDetection();
      }
    }, 4000);
  }, [band, speakWord]);

  const scheduleNextDetection = useCallback(() => {
    if (detectionTimeoutRef.current) {
      clearTimeout(detectionTimeoutRef.current);
    }
    const interval = getRandomInterval();
    detectionTimeoutRef.current = setTimeout(() => {
      triggerDetection();
    }, interval);
  }, [getRandomInterval, triggerDetection]);

  // ─── SCAN CONTROL ──────────────────────────────────────────────────────────

  const startScan = useCallback(() => {
    isScanningRef.current = true;
    setScanState("scanning");
    setSignalStrength(0);
    setElapsedTime(0);
    setCurrentWord("");

    // Arka plan cızırtı
    try {
      staticPlayer.loop = true;
      staticPlayer.volume = 0.25;
      staticPlayer.play();
    } catch (e) {}

    if (Platform.OS !== "web") {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);
    }

    // Frekans tarama animasyonu
    const bandConfig = BANDS[band];
    let dir = 1;
    scanIntervalRef.current = setInterval(() => {
      setFrequency((prev) => {
        const speed = bandConfig.step * dir * (0.5 + Math.random() * 2);
        let next = prev + speed;
        if (next >= bandConfig.max) { dir = -1; return bandConfig.max; }
        if (next <= bandConfig.min) { dir = 1; return bandConfig.min; }
        return Math.round(next * 10) / 10;
      });
    }, 120);

    // Süre sayacı
    timerIntervalRef.current = setInterval(() => {
      setElapsedTime((prev) => prev + 1);
    }, 1000);

    // İlk tespit zamanla
    scheduleNextDetection();
  }, [band, scheduleNextDetection]);

  const stopScan = useCallback(() => {
    isScanningRef.current = false;
    setScanState("idle");
    setSignalStrength(0);
    setCurrentWord("");

    // Sesleri durdur
    try { staticPlayer.pause(); } catch (e) {}
    Speech.stop();

    // Interval'leri temizle
    if (scanIntervalRef.current) { clearInterval(scanIntervalRef.current); scanIntervalRef.current = null; }
    if (timerIntervalRef.current) { clearInterval(timerIntervalRef.current); timerIntervalRef.current = null; }
    if (detectionTimeoutRef.current) { clearTimeout(detectionTimeoutRef.current); detectionTimeoutRef.current = null; }

    if (Platform.OS !== "web") {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    }
  }, []);

  // Cleanup
  useEffect(() => {
    return () => {
      isScanningRef.current = false;
      if (scanIntervalRef.current) clearInterval(scanIntervalRef.current);
      if (timerIntervalRef.current) clearInterval(timerIntervalRef.current);
      if (detectionTimeoutRef.current) clearTimeout(detectionTimeoutRef.current);
      try { staticPlayer.pause(); } catch (e) {}
      Speech.stop();
    };
  }, []);

  // ─── MANUAL FREQUENCY ──────────────────────────────────────────────────────

  const adjustFrequency = (direction: 1 | -1) => {
    if (scanState !== "idle") return;
    const bandConfig = BANDS[band];
    setFrequency((prev) => {
      const next = prev + bandConfig.step * direction * 5;
      return Math.max(bandConfig.min, Math.min(bandConfig.max, Math.round(next * 10) / 10));
    });
    if (Platform.OS !== "web") {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    }
  };

  // ─── BAND SWITCH ───────────────────────────────────────────────────────────

  const switchBand = (newBand: Band) => {
    if (scanState !== "idle") return;
    setBand(newBand);
    setFrequency(BANDS[newBand].min);
    if (Platform.OS !== "web") {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    }
  };

  // ─── HELPERS ────────────────────────────────────────────────────────────────

  const formatFreq = (freq: number, b: Band) => {
    return b === "AM" ? `${Math.round(freq)}` : freq.toFixed(1);
  };

  const formatTime = (seconds: number) => {
    const m = Math.floor(seconds / 60);
    const s = seconds % 60;
    return `${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
  };

  const getStrengthColor = (s: number) => {
    if (s > 75) return "#FF4444";
    if (s > 50) return "#FFD700";
    if (s > 25) return "#00FF88";
    return "#4A4A5A";
  };

  // ─── RENDER ─────────────────────────────────────────────────────────────────

  const isActive = scanState !== "idle";
  const isDetected = scanState === "detected";
  const bandConfig = BANDS[band];

  return (
    <ScreenContainer containerClassName="bg-[#060609]">
      <ScrollView
        style={styles.scrollView}
        contentContainerStyle={styles.scrollContent}
        showsVerticalScrollIndicator={false}
      >
        {/* ─── HEADER ─────────────────────────────────────── */}
        <View style={styles.header}>
          <Text style={styles.title}>FREKANS TARAYICI</Text>
          <View style={styles.headerRow}>
            <Text style={styles.subtitle}>Paranormal Sinyal Avcısı</Text>
            {isActive && (
              <Text style={styles.timer}>{formatTime(elapsedTime)}</Text>
            )}
          </View>
        </View>

        {/* ─── BAND SELECTOR ──────────────────────────────── */}
        <View style={styles.bandRow}>
          {(["AM", "FM"] as Band[]).map((b) => (
            <Pressable
              key={b}
              onPress={() => switchBand(b)}
              style={({ pressed }) => [
                styles.bandBtn,
                band === b && styles.bandBtnActive,
                pressed && { opacity: 0.7 },
              ]}
            >
              <Text style={[styles.bandLabel, band === b && styles.bandLabelActive]}>
                {b}
              </Text>
              <Text style={styles.bandInfo}>
                {BANDS[b].min}–{BANDS[b].max} {BANDS[b].unit}
              </Text>
            </Pressable>
          ))}
        </View>

        {/* ─── FREQUENCY DISPLAY ──────────────────────────── */}
        <Animated.View
          style={[
            styles.freqCard,
            { transform: [{ scale: pulseAnim }] },
            isDetected && styles.freqCardDetected,
          ]}
        >
          {/* Glow overlay */}
          <Animated.View
            style={[styles.glowOverlay, { opacity: glowAnim }]}
          />

          <Text style={styles.freqBandLabel}>
            {band} BANT
          </Text>

          <Text
            style={[styles.freqValue, isDetected && styles.freqValueDetected]}
            numberOfLines={1}
            adjustsFontSizeToFit
          >
            {formatFreq(frequency, band)}
          </Text>

          <Text style={styles.freqUnit}>{bandConfig.unit}</Text>

          {/* Sinyal gücü */}
          {signalStrength > 0 && (
            <View style={styles.strengthRow}>
              <View style={styles.strengthBarBg}>
                {Array.from({ length: 10 }).map((_, i) => (
                  <View
                    key={i}
                    style={[
                      styles.strengthSegment,
                      {
                        backgroundColor:
                          i < Math.ceil(signalStrength / 10)
                            ? getStrengthColor(signalStrength)
                            : "#1A1A2E",
                      },
                    ]}
                  />
                ))}
              </View>
              <Text style={[styles.strengthText, { color: getStrengthColor(signalStrength) }]}>
                {signalStrength}%
              </Text>
            </View>
          )}

          {/* Durum */}
          <View style={styles.statusRow}>
            <View
              style={[
                styles.statusDot,
                {
                  backgroundColor: isDetected ? "#9B4FDE" : isActive ? "#00FF88" : "#333",
                },
              ]}
            />
            <Text style={[styles.statusText, isDetected && { color: "#9B4FDE" }]}>
              {isDetected ? "SİNYAL YAKALANDI" : isActive ? "Taranıyor..." : "Hazır"}
            </Text>
          </View>
        </Animated.View>

        {/* ─── TESPİT EDİLEN KELİME ──────────────────────── */}
        <Animated.View style={[styles.wordContainer, { opacity: wordFadeAnim }]}>
          {currentWord !== "" && (
            <Text style={styles.wordText}>"{currentWord}"</Text>
          )}
        </Animated.View>

        {/* ─── SPECTRUM ───────────────────────────────────── */}
        <Animated.View style={[styles.spectrum, { opacity: noiseAnim }]}>
          {Array.from({ length: 24 }).map((_, i) => {
            const h = isActive ? 3 + Math.random() * (isDetected ? 28 : 14) : 3;
            return (
              <View
                key={i}
                style={[
                  styles.specBar,
                  {
                    height: h,
                    backgroundColor: isDetected
                      ? "#9B4FDE"
                      : "#00FF88",
                    opacity: isDetected ? 0.6 + Math.random() * 0.4 : 0.3 + Math.random() * 0.4,
                  },
                ]}
              />
            );
          })}
        </Animated.View>

        {/* ─── CONTROLS ───────────────────────────────────── */}
        <View style={styles.controlRow}>
          <Pressable
            onPress={() => adjustFrequency(-1)}
            disabled={isActive}
            style={({ pressed }) => [
              styles.arrowBtn,
              pressed && { opacity: 0.5 },
              isActive && { opacity: 0.2 },
            ]}
          >
            <Text style={styles.arrowText}>−</Text>
          </Pressable>

          <Pressable
            onPress={isActive ? stopScan : startScan}
            style={({ pressed }) => [
              styles.mainBtn,
              isActive && styles.mainBtnActive,
              pressed && { transform: [{ scale: 0.96 }] },
            ]}
          >
            <Text
              style={[styles.mainBtnIcon, isActive && styles.mainBtnIconActive]}
            >
              {isActive ? "■" : "▶"}
            </Text>
            <Text
              style={[styles.mainBtnLabel, isActive && styles.mainBtnLabelActive]}
              numberOfLines={1}
            >
              {isActive ? "DURDUR" : "BAŞLAT"}
            </Text>
          </Pressable>

          <Pressable
            onPress={() => adjustFrequency(1)}
            disabled={isActive}
            style={({ pressed }) => [
              styles.arrowBtn,
              pressed && { opacity: 0.5 },
              isActive && { opacity: 0.2 },
            ]}
          >
            <Text style={styles.arrowText}>+</Text>
          </Pressable>
        </View>

        {/* ─── DETECTION HISTORY ──────────────────────────── */}
        {detections.length > 0 && (
          <View style={styles.historyCard}>
            <Text style={styles.historyTitle}>Yakalanan Sinyaller</Text>
            {detections.map((d) => (
              <View key={d.id} style={styles.historyRow}>
                <View style={styles.historyLeft}>
                  <Text style={styles.historyWord}>"{d.word}"</Text>
                  <Text style={styles.historyMeta}>
                    {formatFreq(d.frequency, d.band)} {BANDS[d.band].unit}
                  </Text>
                </View>
                <View style={[styles.historyBadge, { backgroundColor: getStrengthColor(d.strength) }]}>
                  <Text style={styles.historyBadgeText}>{d.strength}%</Text>
                </View>
              </View>
            ))}
          </View>
        )}

        {/* Alt boşluk (tab bar için) */}
        <View style={{ height: 100 }} />
      </ScrollView>
    </ScreenContainer>
  );
}
